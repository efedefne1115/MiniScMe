function timeAgo(ts){
  if(!ts) return '';
  const d = ts.toDate ? ts.toDate() : new Date(ts);
  const s = Math.floor((Date.now() - d.getTime())/1000);
  if(s < 60) return 'şimdi';
  if(s < 3600) return Math.floor(s/60)+'dk';
  if(s < 86400) return Math.floor(s/3600)+'sa';
  if(s < 604800) return Math.floor(s/86400)+'g';
  return d.toLocaleDateString('tr-TR');
}

  if(c === 'auth/too-many-requests') return 'Çok fazla deneme yapıldı, biraz sonra tekrar dene.';

}

      state.uid = user.uid;
      const doc = await db.collection('users').doc(user.uid).get();
      if(!doc.exists){ await auth.signOut(); return; }
      state.profile = doc.data();
      enterApp();
    }else{
      teardownListeners();
      state.uid = null; state.profile = null;
      document.getElementById('appScreen').classList.add('hidden');
      document.getElementById('authScreen').classList.remove('hidden');
      document.getElementById('loginForm').reset();
      document.getElementById('signupForm').reset();
    }
  });
}

async function startChatWith(otherUid, otherUsername){
  if(otherUid === state.uid) return;
  const myBlocked = state.profile.blocked || [];
  if(myBlocked.includes(otherUid)){ toast('Bu kullanıcıyı engelledin.'); return; }
  const otherDoc = await db.collection('users').doc(otherUid).get();
  const otherData = otherDoc.data() || {};
  if((otherData.blocked||[]).includes(state.uid)){ toast('Bu kullanıcıya mesaj gönderemezsin.'); return; }
  if(otherData.allowMessages === false){ toast('Bu kullanıcı yeni mesajlara kapalı.'); return; }

  const cid = chatIdFor(state.uid, otherUid);
  const ref = db.collection('chats').doc(cid);
  const doc = await ref.get();
  if(!doc.exists){
    await ref.set({
      participants: [state.uid, otherUid],
      usernames: { [state.uid]: state.profile.username, [otherUid]: otherUsername },
      status: 'pending',
      requestedBy: state.uid,
      lastMessage: '',
      updatedAt: firebase.firestore.FieldValue.serverTimestamp()
    });
  } else if(doc.data().status === 'rejected' && doc.data().requestedBy !== state.uid){
    // reddedilmiş ama karşı taraf tekrar yazıyor -> yeniden istek haline getir
    await ref.update({ status:'pending', requestedBy: state.uid, updatedAt: firebase.firestore.FieldValue.serverTimestamp() });
  }
  document.getElementById('userSearchInput').value = '';
  document.getElementById('userSearchResults').classList.add('hidden');
  openChat(cid, otherUid, otherUsername);
}

function attachMyChats(){
  if(state.unsub.chats) return;
  state.unsub.chats = db.collection('chats').where('participants','array-contains', state.uid)
    .onSnapshot(snap=>{
      const requests = [];
      const accepted = [];
      snap.forEach(doc=>{
        const c = doc.data();
        const otherUid = c.participants.find(p=>p!==state.uid);
        if(!otherUid) return;
        const otherName = (c.usernames && c.usernames[otherUid]) || 'kullanıcı';
        const item = { id: doc.id, otherUid, otherName, ...c };
        if(c.status === 'pending'){
          if(c.requestedBy !== state.uid) requests.push(item); // gelen istek
        } else if(c.status === 'accepted'){
          accepted.push(item);
        }
      });
      accepted.sort((a,b)=> (b.updatedAt?.toMillis?.()||0) - (a.updatedAt?.toMillis?.()||0));

      document.getElementById('msgNavBadge').textContent = requests.length;
      document.getElementById('msgNavBadge').classList.toggle('show', requests.length>0);

      const reqSection = document.getElementById('requestsSection');
      reqSection.classList.toggle('hidden', requests.length===0);
      document.getElementById('requestsList').innerHTML = requests.map(r=>`
        <div class="req-card">
          <div class="req-top">
            <div class="avatar sm">${esc(initials(r.otherName))}</div>
            <div class="meta"><div class="uname">${esc(r.otherName)}</div><div class="usub">sana mesaj göndermek istiyor</div></div>
          </div>
          <div class="req-actions">
            <button class="btn btn-ghost btn-sm reject-req" data-id="${r.id}">Reddet</button>
            <button class="btn btn-primary btn-sm accept-req" data-id="${r.id}" data-uid="${r.otherUid}" data-name="${esc(r.otherName)}">Kabul Et</button>
          </div>
        </div>`).join('');

      const list = document.getElementById('chatsList');
      if(accepted.length===0){
        list.innerHTML = '<div class="sub" style="color:var(--text-faint);font-size:13px;padding:6px 4px;">Henüz sohbetin yok.</div>';
      }else{
        list.innerHTML = accepted.map(c=>`
          <div class="chat-row" data-id="${c.id}" data-uid="${c.otherUid}" data-name="${esc(c.otherName)}">
            <div class="avatar">${esc(initials(c.otherName))}</div>
            <div class="meta"><div class="uname">${esc(c.otherName)}</div><div class="lastmsg">${esc(c.lastMessage||'Sohbete başla')}</div></div>
            <div class="time">${timeAgo(c.updatedAt)}</div>
          </div>`).join('');
      }
    });
}

async function requestJoinRoom(roomId, room){
  if((room.members||[]).length >= room.maxMembers){ toast('Oda dolu.'); return; }
  const ref = db.collection('rooms').doc(roomId);
  if(room.isPublic){
    await ref.update({
      members: firebase.firestore.FieldValue.arrayUnion(state.uid),
      ['memberUsernames.'+state.uid]: state.profile.username
    });
    toast('Odaya katıldın.');
    openRoom(roomId);
  }else{
    if((room.joinRequests||[]).includes(state.uid)){ toast('İsteğin zaten gönderildi.'); return; }
    await ref.update({ joinRequests: firebase.firestore.FieldValue.arrayUnion(state.uid) });
    toast('Katılma isteği gönderildi.');
  }
}

async function openRoom(roomId){
  const doc = await db.collection('rooms').doc(roomId).get();
  if(!doc.exists){ toast('Oda bulunamadı.'); return; }
  state.activeRoomId = roomId;
  state.activeRoom = { id: roomId, ...doc.data() };
  document.getElementById('roomsListView').classList.add('hidden');
  document.getElementById('roomDetailView').classList.remove('hidden');
  refreshRoomHeader(state.activeRoom);

  if(state.unsub.roomMeta) state.unsub.roomMeta();
  state.unsub.roomMeta = db.collection('rooms').doc(roomId).onSnapshot(d=>{
    if(!d.exists){ toast('Oda silindi.'); closeRoom(); return; }
    state.activeRoom = { id: roomId, ...d.data() };
    refreshRoomHeader(state.activeRoom);
  });

  if(state.unsub.roomMsgs) state.unsub.roomMsgs();
  state.unsub.roomMsgs = db.collection('rooms').doc(roomId).collection('messages').orderBy('createdAt','asc').limit(300)
    .onSnapshot(snap=>{
      const body = document.getElementById('roomBody');
      body.innerHTML = '';
      snap.forEach(d=>{
        const m = d.data();
        const mine = m.uid === state.uid;
        const b = document.createElement('div');
        b.className = 'msg-bubble ' + (mine?'mine':'theirs');
        b.innerHTML = (mine?'':`<span class="msg-author">${esc(m.username)}</span>`) + esc(m.text);
        body.appendChild(b);
      });
      body.scrollTop = body.scrollHeight;
    });
}

function refreshRoomHeader(room){
  document.getElementById('roomTitleText').textContent = room.name;
  document.getElementById('roomMemberCountText').textContent = (room.members||[]).length + '/' + room.maxMembers + ' üye';
  const isOwner = room.ownerUid === state.uid;
  document.getElementById('roomRequestsBtn').classList.toggle('hidden', !isOwner || room.isPublic);
  const reqCount = (room.joinRequests||[]).length;
  document.getElementById('roomRequestsBtn').style.position='relative';
}

function closeRoom(){
  if(state.unsub.roomMsgs){ state.unsub.roomMsgs(); delete state.unsub.roomMsgs; }
  if(state.unsub.roomMeta){ state.unsub.roomMeta(); delete state.unsub.roomMeta; }
  state.activeRoomId = null; state.activeRoom = null;
  document.getElementById('roomDetailView').classList.add('hidden');
  document.getElementById('roomsListView').classList.remove('hidden');
}
document.getElementById('backFromRoomBtn').onclick = closeRoom;

async function sendRoomMessage(){
  const input = document.getElementById('roomMsgInput');
  const text = input.value.trim();
  if(!text || !state.activeRoomId) return;
  input.value = '';
  try{
    await db.collection('rooms').doc(state.activeRoomId).collection('messages').add({
      uid: state.uid, username: state.profile.username, text, createdAt: firebase.firestore.FieldValue.serverTimestamp()
    });
  }catch(e){ toast('Gönderilemedi.'); }
}
document.getElementById('roomSendBtn').onclick = sendRoomMessage;
document.getElementById('roomMsgInput').addEventListener('keydown', e=>{ if(e.key==='Enter') sendRoomMessage(); });

document.getElementById('roomRequestsBtn').onclick = ()=>{
  if(!state.activeRoom) return;
  const room = state.activeRoom;
  const reqs = room.joinRequests || [];
  const root = document.getElementById('modalRoot');
  root.innerHTML = `
    <div class="modal-overlay" id="reqOverlay">
      <div class="modal-card">
        <div class="modal-title">Katılma İstekleri</div>
        <div id="roomReqList">${reqs.length===0?'<div class="sub" style="color:var(--text-faint);font-size:13px;">Bekleyen istek yok.</div>':''}</div>
      </div>
    </div>`;
  document.getElementById('reqOverlay').onclick = (e)=>{ if(e.target.id==='reqOverlay') closeModal(); };
  const listEl = document.getElementById('roomReqList');
  reqs.forEach(async (uid)=>{
    const udoc = await db.collection('users').doc(uid).get();
    const uname = udoc.exists ? udoc.data().username : 'kullanıcı';
    const row = document.createElement('div');
    row.className = 'req-card';
    row.innerHTML = `
      <div class="req-top"><div class="avatar sm">${esc(initials(uname))}</div><div class="meta"><div class="uname">${esc(uname)}</div><div class="usub">katılmak istiyor</div></div></div>
      <div class="req-actions">
        <button class="btn btn-ghost btn-sm decline-join">Reddet</button>
        <button class="btn btn-primary btn-sm accept-join">Kabul Et</button>
      </div>`;
    row.querySelector('.accept-join').onclick = async ()=>{
      await db.collection('rooms').doc(room.id).update({
        joinRequests: firebase.firestore.FieldValue.arrayRemove(uid),
        members: firebase.firestore.FieldValue.arrayUnion(uid),
        ['memberUsernames.'+uid]: uname
      });
      row.remove();
      toast(uname + ' odaya katıldı.');
    };
    row.querySelector('.decline-join').onclick = async ()=>{
      await db.collection('rooms').doc(room.id).update({ joinRequests: firebase.firestore.FieldValue.arrayRemove(uid) });
      row.remove();
    };
    listEl.appendChild(row);
  });
};

document.getElementById('roomLeaveOrDeleteBtn').onclick = ()=>{
  if(!state.activeRoom) return;
  const room = state.activeRoom;
  const isOwner = room.ownerUid === state.uid;
  if(isOwner){
    showConfirm('Odayı Sil', '"' + room.name + '" odası kalıcı olarak silinecek. Emin misin?', 'Odayı Sil', async ()=>{
      await db.collection('rooms').doc(room.id).delete();
      toast('Oda silindi.');
      closeRoom();
    }, true);
  }else{
    showConfirm('Odadan Ayrıl', '"' + room.name + '" odasından ayrılmak istediğine emin misin?', 'Ayrıl', async ()=>{
      await db.collection('rooms').doc(room.id).update({
        members: firebase.firestore.FieldValue.arrayRemove(state.uid)
      });
      closeRoom();
    }, true);
  }
};

function renderBlockedList(){
  const box = document.getElementById('blockedList');
  const blocked = state.profile.blocked || [];
  if(blocked.length === 0){ box.innerHTML = '<div class="sub">Henüz kimseyi engellemedin.</div>'; return; }
  box.innerHTML = '';
  blocked.forEach(async (uid)=>{
    const udoc = await db.collection('users').doc(uid).get();
    const uname = udoc.exists ? udoc.data().username : 'kullanıcı';
    const row = document.createElement('div');
    row.className = 'blocked-row';
    row.innerHTML = `<div class="avatar sm">${esc(initials(uname))}</div><div class="meta">${esc(uname)}</div><button class="btn btn-ghost btn-sm">Kaldır</button>`;
    row.querySelector('button').onclick = async ()=>{
      await db.collection('users').doc(state.uid).update({ blocked: firebase.firestore.FieldValue.arrayRemove(uid) });
      state.profile.blocked = (state.profile.blocked||[]).filter(x=>x!==uid);
      row.remove();
      if((state.profile.blocked||[]).length===0) box.innerHTML = '<div class="sub">Henüz kimseyi engellemedin.</div>';
      toast(uname + ' engeli kaldırıldı.');
    };
    box.appendChild(row);
  });
}
